cmake_minimum_required(VERSION 3.16)
project(firmware_host_tests C)
enable_testing()
set(CMAKE_C_STANDARD 11)
option(ENABLE_SANITIZERS "启用地址和未定义行为检查" OFF)
if(NOT IDF_PATH)
    message(FATAL_ERROR "请通过 -DIDF_PATH 指定 ESP-IDF 路径")
endif()
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_library(cjson STATIC "${IDF_PATH}/components/json/cJSON/cJSON.c")
target_include_directories(cjson PUBLIC "${IDF_PATH}/components/json/cJSON")
add_library(models STATIC
    "${ROOT}/lib/app_config/config_model.c"
    "${ROOT}/lib/content/content_item.c"
    "${ROOT}/lib/app_controller/app_state.c")
target_include_directories(models PUBLIC "${ROOT}/lib/app_config" "${ROOT}/lib/content" "${ROOT}/lib/app_controller")
target_link_libraries(models PUBLIC cjson m)
target_compile_options(models PRIVATE -Wall -Wextra -Werror)
if(ENABLE_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
    target_compile_options(models PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_compile_options(cjson PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
add_executable(test_models test_models.c)
target_link_libraries(test_models PRIVATE models)
target_compile_options(test_models PRIVATE -Wall -Wextra -Werror)
add_test(NAME models COMMAND test_models)
add_executable(test_transport test_transport.c
    "${ROOT}/lib/http_transport/http_transport.c"
    "${ROOT}/lib/backend_client/backend_client.c")
target_include_directories(test_transport PRIVATE stubs "${ROOT}/lib/http_transport" "${ROOT}/lib/backend_client")
target_compile_options(test_transport PRIVATE -Wall -Wextra -Werror)
target_link_libraries(test_transport PRIVATE models)
add_test(NAME transport COMMAND test_transport)
add_executable(test_config_store test_config_store.c "${ROOT}/lib/app_config/app_config.c")
target_include_directories(test_config_store PRIVATE stubs "${ROOT}/lib/nvs_manager")
target_compile_options(test_config_store PRIVATE -Wall -Wextra -Werror)
target_link_libraries(test_config_store PRIVATE models)
foreach(scenario defaults migration custom saved corrupt read-failure)
    add_test(NAME "config_${scenario}" COMMAND test_config_store "${scenario}")
endforeach()
